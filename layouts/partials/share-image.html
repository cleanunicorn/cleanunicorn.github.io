{{- /* The page's social share image, as a dict: url, and width and height when known. Used by head.html (og:image, twitter:image) and structured_data.html (BlogPosting image), so all three always agree. Order: 1. front-matter `cover`: a bundle resource, else a static path or URL (used as given, with no dimensions, since nothing is processed; so is a bundle cover that is not a raster image, such as an SVG); 2. the first markdown image in the content that is a bundle resource — the post's lead image (resources sort by name, so "image 1.png" would come before "image.png"); 3. the first image resource in the bundle; 4. the site-wide og-image.png. A raster resource from 1-3 is cropped to 1200x630 with Hugo image processing. */ -}} {{- $result := dict "url" ("og-image.png" | absURL) "width" 1200 "height" 630 -}} {{- $res := false -}} {{- $static := false -}} {{- with .Params.cover -}} {{- with $.Resources.GetMatch . -}} {{- $res = . -}} {{- else -}} {{- $static = true -}} {{- $result = dict "url" (. | absURL) -}} {{- end -}} {{- end -}} {{- if not (or $res $static) -}} {{- range findRESubmatch `!\[[^\]]*\]\(\s*]+)` .RawContent -}} {{- if not $res -}} {{- with (urls.Parse (index . 1)).Path -}} {{- with $.Resources.GetMatch (strings.TrimPrefix "./" .) -}} {{- if and (eq .ResourceType "image") (ne .MediaType.SubType "svg") -}} {{- $res = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if not (or $res $static) -}} {{- range .Resources.ByType "image" -}} {{- if and (not $res) (ne .MediaType.SubType "svg") -}} {{- $res = . -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Only raster images can be processed. A cover that is an SVG (or any other non-raster resource) is used as given, with no dimensions, like a static-path cover; .Fill on it would fail the build. */ -}} {{- with $res -}} {{- if and (eq .ResourceType "image") (ne .MediaType.SubType "svg") -}} {{- with .Fill "1200x630 Center" -}} {{- $result = dict "url" .Permalink "width" .Width "height" .Height -}} {{- end -}} {{- else -}} {{- $result = dict "url" .Permalink -}} {{- end -}} {{- end -}} {{- return $result -}}
