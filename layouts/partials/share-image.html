{{- /* The page's social share image, as a dict: url, and width and height when known. Used by head.html (og:image, twitter:image) and structured_data.html (BlogPosting image), so all three always agree. Order: 1. front-matter `cover`: a bundle resource, else a static path or URL (used as given, with no dimensions, since nothing is processed; so is a bundle cover that is not a raster image, such as an SVG, because .Fill would fail the build on it); 2. the first markdown image in the content that is a bundle resource — the post's lead image (resources sort by name, so "image 1.png" would come before "image.png"); 3. the first image resource in the bundle; 4. the site-wide og-image.png. A raster resource from 1-3 is cropped to 1200x630 with Hugo image processing. */ -}} {{- $share := dict "url" ("og-image.png" | absURL) "width" 1200 "height" 630 -}} {{- $image := false -}} {{- $coverIsPath := false -}} {{- with .Params.cover -}} {{- with $.Resources.GetMatch . -}} {{- $image = . -}} {{- else -}} {{- $coverIsPath = true -}} {{- $share = dict "url" (. | absURL) -}} {{- end -}} {{- end -}} {{- if not (or $image $coverIsPath) -}} {{- range findRESubmatch `!\[[^\]]*\]\(\s*]+)` .RawContent -}} {{- with (urls.Parse (index . 1)).Path -}} {{- with $.Resources.GetMatch (strings.TrimPrefix "./" .) -}} {{- if and (eq .ResourceType "image") (ne .MediaType.SubType "svg") -}} {{- $image = . -}} {{- break -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if not (or $image $coverIsPath) -}} {{- range .Resources.ByType "image" -}} {{- if ne .MediaType.SubType "svg" -}} {{- $image = . -}} {{- break -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Only raster images can be processed; see item 1 above. */ -}} {{- with $image -}} {{- if and (eq .ResourceType "image") (ne .MediaType.SubType "svg") -}} {{- with .Fill "1200x630 Center" -}} {{- $share = dict "url" .Permalink "width" .Width "height" .Height -}} {{- end -}} {{- else -}} {{- $share = dict "url" .Permalink -}} {{- end -}} {{- end -}} {{- return $share -}}
