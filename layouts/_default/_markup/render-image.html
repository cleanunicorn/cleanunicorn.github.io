{{- /* Post images: serve a WebP copy (max 1600px wide, never upscaled) with its dimensions and lazy loading. GIF/SVG pass through unchanged. The build fails on a served image over 300 KB or a page-relative path that is not a resource; remote, data: and root-relative destinations are emitted as-is. Resolution mirrors Hugo's embedded image hook. */ -}} {{- $src := .Destination -}} {{- $w := 0 -}}{{- $h := 0 -}} {{- $destURL := urls.Parse .Destination -}} {{- if not $destURL.IsAbs -}} {{- $path := strings.TrimPrefix "./" $destURL.Path -}} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $img := . -}} {{- $raster := and (eq .ResourceType "image") (not (in (slice "gif" "svg") .MediaType.SubType)) -}} {{- if and $raster (gt .Width 1600) -}} {{- $img = .Resize "1600x webp q80 Lanczos" -}} {{- else if $raster -}} {{- $img = .Process "webp q80 Lanczos" -}} {{- end -}} {{- if gt (len $img.Content) 307200 -}} {{- $msg := printf "%s: image %q is %d bytes as served; post images must be at most 300 KB (GIF/SVG are not resized)" $.Page.File.Path $.Destination (len $img.Content) -}} {{- if $.Page.Draft -}}{{- warnf "%s" $msg -}}{{- else -}}{{- errorf "%s" $msg -}}{{- end -}} {{- end -}} {{- $src = $img.RelPermalink -}} {{- if and (eq $img.ResourceType "image") (ne $img.MediaType.SubType "svg") -}} {{- $w = $img.Width -}}{{- $h = $img.Height -}} {{- end -}} {{- else -}} {{- if not (or $destURL.Host (strings.HasPrefix $destURL.Path "/")) -}} {{- errorf "%s: image %q is not a resource of this page (check the file name)" .Page.File.Path .Destination -}} {{- end -}} {{- end -}} {{- end -}} {{- /* no trailing newline */ -}}
